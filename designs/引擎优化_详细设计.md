## 二、F6: DAG 原子写入（v2.0 修正：UUID + Windows降级 + 排他锁 + 乐观锁）

### 2.1 修改方案

在 `tools/shared.ts` 新增 `atomicWriteJSON()` 函数：

```typescript
import { randomUUID } from "node:crypto";

/**
 * F6: 原子写入 JSON 文件
 * 三层保护:
 *   1. UUID 命名 → 避免 PID+timestamp 预测攻击
 *   2. backup-then-replace → Windows rename 非原子覆盖的降级路径
 *   3. 排他锁 → 防止多进程并发写入
 */
export function atomicWriteJSON<T>(
  filePath: string,
  data: T,
  options?: { expectedVersion?: number }
): void {
  const tmpPath = `${filePath}.${randomUUID()}.tmp`;
  const lockPath = `${filePath}.lock`;
  const backupPath = `${filePath}.bak`;

  // 1. 获取排他文件锁
  const lockFd = fs.openSync(lockPath, "wx");  // 'wx'=排他创建
  try {
    // 2. 乐观锁检查（可选，用于 F8）
    if (options?.expectedVersion !== undefined && fs.existsSync(filePath)) {
      const current = JSON.parse(fs.readFileSync(filePath, "utf-8"));
      if (current._version !== options.expectedVersion) {
        throw new Error(`Version conflict: expected ${options.expectedVersion}`);
      }
    }

    // 3. 写临时文件 + fsync 强制刷盘
    const fd = fs.openSync(tmpPath, "w");
    fs.writeFileSync(fd, JSON.stringify(data, null, 2), "utf-8");
    fs.fsyncSync(fd);
    fs.closeSync(fd);

    // 4. 跨平台原子替换（含 Windows 降级路径）
    if (fs.existsSync(filePath)) {
      try {
        fs.renameSync(tmpPath, filePath);      // POSIX: 原子覆盖
      } catch {
        // Windows 降级：备份 → 删除 → 重命名
        if (fs.existsSync(backupPath)) fs.unlinkSync(backupPath);
        fs.renameSync(filePath, backupPath);
        try {
          fs.renameSync(tmpPath, filePath);
          fs.unlinkSync(backupPath);
        } catch (err2) {
          fs.renameSync(backupPath, filePath);  // 回滚
          throw new Error(`Atomic write failed, rolled back: ${err2}`);
        }
      }
    } else {
      fs.renameSync(tmpPath, filePath);
    }
  } finally {
    fs.closeSync(lockFd);
    try { fs.unlinkSync(lockPath); } catch {}
    try { if (fs.existsSync(tmpPath)) fs.unlinkSync(tmpPath); } catch {}
  }
}
```

### 2.2 孤儿 tmp 文件清理（启动时执行一次）

```typescript
export function cleanupOrphanedTmpFiles(dirPath: string): void {
  for (const entry of fs.readdirSync(dirPath, { withFileTypes: true })) {
    if (entry.isFile() && entry.name.endsWith(".tmp")) {
      const age = Date.now() - fs.statSync(path.join(dirPath, entry.name)).mtimeMs;
      if (age > 60_000) {
        try { fs.unlinkSync(path.join(dirPath, entry.name)); } catch {}
      }
    }
  }
}
```
